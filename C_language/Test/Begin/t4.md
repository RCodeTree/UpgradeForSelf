/* 
正整数 A 和正整数 B 的最小公倍数是指能被 A 和 B 整除的最小的正整数值，设计一个算法，求输入 A 和 B 的最小公倍数。

输入描述：
输入两个正整数 A 和 B。

输出描述：
输出 A 和 B 的最小公倍数。
*/
#include <stdio.h>

/* 
方法一：使用两个数中较大的那个数作为最小公倍数的初始值，每次增加 1 ，直到找到一个数能同时整除 num1 和 num2 ，则该数为最小公倍数
*/
int gcd1(int num1, int num2)
{
    int max = num1 > num2 ? num1 : num2;
    while(1)
    {
            if(max % num1 == 0 && max % num2 == 0)
            {
                break;
            }

            max++;
    }
    return max;
}

/* 
方法二：使用两个数中的其中一个数，使用一个数从 0 开始，每次乘这个数，如果乘积能被另一个数整除，则该乘积为最小公倍数
*/
int gcd2(int num1, int num2)
{
   int i = 0;
   while(1)
   {
        if(num1 * i % num2 == 0)
        {
            break;
        }

        i++;
   }

   return num1 * i;
}


int main()
{
    unsigned int num1 = 0;
    unsigned int num2 = 0;

    prinft("请输入两个正整数: ");
    scanf("%d%d", &num1, &num2);

    int lcm = gcd1(num1, num2);
    printf("%d 和 %d 的最小公倍数为: %d\n", num1, num2, lcm);

    return 0;
}
