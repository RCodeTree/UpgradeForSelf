// 以下程序的输出结果？(实际考察递归调用的次数)
/* 
做题技巧：
将每一次的递归的数字一一拆开，最终拆开会有多少个数字就是递归的次数
8
7 6
6 5 5 4 
........
*/
#include <stdio.h>
int cnt = 0;
int fib(int n)
{
    cnt++;
    if(n == 0)
    {
        return 1;
    }
    else if(n == 1)
    {
        return 2;
    }
    else
    {
        return fib(n - 1) + fib(n - 2);
    } 
}

int main()
{
   fib(8);
   printf("%d", cnt); // 67
}
